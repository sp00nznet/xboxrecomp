cmake_minimum_required(VERSION 3.20)
project(nv2a_pvideo_test C)
include(${CMAKE_CURRENT_LIST_DIR}/../xr_test.cmake)
if(NOT WIN32)
    xr_test_refuse("The PVIDEO hook is Windows-only")
    return()
endif()
set(CMAKE_C_STANDARD 11)
# Under tests/CMakeLists.txt the whole suite shares one runtime build.
if(NOT TARGET xboxrecomp)
    add_subdirectory(../.. runtime)
endif()
add_executable(nv2a_pvideo_test test_main.c)
target_include_directories(nv2a_pvideo_test PRIVATE ../../src ../../src/nv2a)
target_link_libraries(nv2a_pvideo_test PRIVATE xboxrecomp)
enable_testing()
add_test(NAME nv2a_pvideo COMMAND nv2a_pvideo_test)
