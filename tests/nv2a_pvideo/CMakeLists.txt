cmake_minimum_required(VERSION 3.20)
project(nv2a_pvideo_test C)
if(NOT WIN32)
    message(FATAL_ERROR "The PVIDEO hook is Windows-only")
endif()
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)

target_include_directories(nv2a_pvideo_test PRIVATE ../../src ../../src/nv2a)
target_link_libraries(nv2a_pvideo_test PRIVATE xboxrecomp)
enable_testing()
add_test(NAME nv2a_pvideo COMMAND nv2a_pvideo_test)
