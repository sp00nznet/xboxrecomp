cmake_minimum_required(VERSION 3.20)
project(x87_trig_regression C)
enable_testing()
add_executable(x87-trig-test x87_trig_test.c)
target_include_directories(x87-trig-test PRIVATE ../../templates/runtime ../../src)
if(NOT WIN32)
    target_link_libraries(x87-trig-test PRIVATE m)
endif()
add_test(NAME guest-x87-trig-pi66 COMMAND x87-trig-test)
# The negative control: the same test against the old lifting (plain libm)
# must fail, or the vectors no longer tell the two apart.
add_executable(x87-trig-test-plain-libm x87_trig_test.c)
target_compile_definitions(x87-trig-test-plain-libm PRIVATE X87_TRIG_PLAIN_LIBM)
target_include_directories(x87-trig-test-plain-libm PRIVATE ../../templates/runtime ../../src)
if(NOT WIN32)
    target_link_libraries(x87-trig-test-plain-libm PRIVATE m)
endif()
add_test(NAME guest-x87-trig-plain-libm-fails COMMAND x87-trig-test-plain-libm)
set_tests_properties(guest-x87-trig-plain-libm-fails PROPERTIES WILL_FAIL TRUE)
