cmake_minimum_required(VERSION 3.20)
project(nv2a_combiner_regression C)
enable_testing()
add_executable(nv2a-combiner-test nv2a_combiner_test.c ../../src/kernel/nv2a_combiner.c)
target_include_directories(nv2a-combiner-test PRIVATE ../../src/kernel)
if(NOT WIN32)
    target_link_libraries(nv2a-combiner-test PRIVATE m)
endif()
add_test(NAME nv2a-register-combiners COMMAND nv2a-combiner-test)

add_executable(nv2a-combiner-fast-test nv2a_combiner_fast_test.c ../../src/kernel/nv2a_combiner.c)
target_include_directories(nv2a-combiner-fast-test PRIVATE ../../src/kernel)
target_compile_definitions(nv2a-combiner-fast-test PRIVATE NV2A_RC_TEST)
if(NOT WIN32)
    target_link_libraries(nv2a-combiner-fast-test PRIVATE m)
endif()
add_test(NAME nv2a-register-combiner-fast-equivalence COMMAND nv2a-combiner-fast-test)
