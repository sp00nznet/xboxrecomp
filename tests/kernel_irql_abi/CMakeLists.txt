cmake_minimum_required(VERSION 3.20)
project(kernel_irql_abi_test C)
include(${CMAKE_CURRENT_LIST_DIR}/../xr_test.cmake)
if(NOT WIN32)
    xr_test_refuse("This synthetic-memory regression uses Windows VirtualAlloc")
    return()
endif()
set(CMAKE_C_STANDARD 11)
# Under tests/CMakeLists.txt the whole suite shares one runtime build.
if(NOT TARGET xboxrecomp)
    add_subdirectory(../.. runtime)
endif()
add_executable(kernel_irql_abi_test test_main.c)
target_include_directories(kernel_irql_abi_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_irql_abi_test PRIVATE xboxrecomp)
enable_testing()
add_test(NAME kernel_irql_abi_test COMMAND kernel_irql_abi_test)
