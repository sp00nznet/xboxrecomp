cmake_minimum_required(VERSION 3.20)
project(kernel_directory_test C)
include(${CMAKE_CURRENT_LIST_DIR}/../xr_test.cmake)
if(NOT WIN32)
    xr_test_refuse("This harness uses Win32 directory handles (FILE_FLAG_BACKUP_SEMANTICS); run it on Windows or under Proton.")
    return()
endif()
set(CMAKE_C_STANDARD 11)
# Under tests/CMakeLists.txt the whole suite shares one runtime build.
if(NOT TARGET xboxrecomp)
    add_subdirectory(../.. runtime)
endif()
add_executable(kernel_directory_test test_main.c)
target_include_directories(kernel_directory_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_directory_test PRIVATE xboxrecomp)
enable_testing()
add_test(NAME kernel_directory_test COMMAND kernel_directory_test)
