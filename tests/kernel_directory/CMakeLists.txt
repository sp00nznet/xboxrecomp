cmake_minimum_required(VERSION 3.20)
project(kernel_directory_test C)
if(NOT WIN32)
    message(FATAL_ERROR "This harness uses Win32 directory handles (FILE_FLAG_BACKUP_SEMANTICS); run it on Windows or under Proton.")
endif()
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
add_executable(kernel_directory_test test_main.c)
target_include_directories(kernel_directory_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_directory_test PRIVATE xboxrecomp)
enable_testing()
add_test(NAME kernel_directory_test COMMAND kernel_directory_test)
