cmake_minimum_required(VERSION 3.20)
project(xaudio2_test C)
if(NOT WIN32)
    message(FATAL_ERROR "This harness drives XAudio2; run it on Windows or under Proton.")
endif()
set(CMAKE_C_STANDARD 11)
add_executable(xaudio2_test test_main.c)
target_include_directories(xaudio2_test PRIVATE ../../src/platform ../../src)
enable_testing()
add_test(NAME xaudio2_test COMMAND xaudio2_test)
