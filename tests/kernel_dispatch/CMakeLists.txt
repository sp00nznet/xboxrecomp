cmake_minimum_required(VERSION 3.20)
project(kernel_dispatch_regression LANGUAGES C)
include(${CMAKE_CURRENT_LIST_DIR}/../xr_test.cmake)
if(NOT WIN32)
    xr_test_refuse("This regression harness currently targets Windows.")
    return()
endif()
# Under tests/CMakeLists.txt the whole suite shares one runtime build.
if(NOT TARGET xboxrecomp)
    add_subdirectory(../.. xboxrecomp)
endif()
add_executable(kernel-dispatch-thread-test kernel_dispatch_thread_test.c)
target_include_directories(kernel-dispatch-thread-test PRIVATE ../../src/kernel)
target_link_libraries(kernel-dispatch-thread-test PRIVATE xboxrecomp)
enable_testing()
add_test(NAME kernel-dispatch-thread COMMAND kernel-dispatch-thread-test)
